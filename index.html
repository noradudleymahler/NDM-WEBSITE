// ── Lightbox with prev/next ─────────────────────────────────────────────── let lightboxImages = []; // list of src strings for the current page/section let lightboxIndex = 0; // index of the currently shown image function buildImageList(clickedSrc) { // Collect all visible images from .gallery-item and .thumbnail-item const imgs = []; document.querySelectorAll( '.page.active .gallery-item img, ' + '.subfolder-view.open .thumbnail-item img' ).forEach(img => imgs.push(img.src)); return imgs; } function openLightbox(src) { lightboxImages = buildImageList(src); lightboxIndex = lightboxImages.indexOf(src); if (lightboxIndex === -1) lightboxIndex = 0; document.getElementById('lightbox-img').src = src; document.getElementById('lightbox').classList.add('open'); } function closeLightbox() { document.getElementById('lightbox').classList.remove('open'); document.getElementById('lightbox-img').src = ''; } function lightboxStep(dir) { if (!lightboxImages.length) return; lightboxIndex = (lightboxIndex + dir + lightboxImages.length) % lightboxImages.length; document.getElementById('lightbox-img').src = lightboxImages[lightboxIndex]; } // Click handler for standard .gallery-item images document.querySelectorAll('.gallery-item').forEach(function (item) { item.addEventListener('click', function () { var img = item.querySelector('img'); if (img) openLightbox(img.src); }); }); // Keyboard: Escape closes, arrows navigate document.addEventListener('keydown', function (e) { if (!document.getElementById('lightbox').classList.contains('open')) return; if (e.key === 'Escape') closeLightbox(); if (e.key === 'ArrowLeft') lightboxStep(-1); if (e.key === 'ArrowRight') lightboxStep(1); });
